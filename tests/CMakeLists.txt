cmake_minimum_required(VERSION 3.20)
# Also runnable alone: this checkout lacks several unrelated upstream test sources.
# When included from the parent, these targets are already registered there.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(strata_cache_tests LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  enable_testing()
  find_package(Threads REQUIRED)
  set(ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
  add_executable(conversation_store_test conversation_store_test.cpp
    "${ROOT}/src/program/conversation_store.cpp" "${ROOT}/src/program/prefix_file.cpp")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(prefix_writeback_test prefix_writeback_test.cpp "${ROOT}/src/program/prefix_file.cpp")
    target_include_directories(prefix_writeback_test PRIVATE "${ROOT}/include")
    target_link_libraries(prefix_writeback_test PRIVATE Threads::Threads)
    target_link_options(prefix_writeback_test PRIVATE "-Wl,--wrap=fsync" "-Wl,--wrap=fdatasync" "-Wl,--wrap=fclose")
    add_test(NAME prefix_writeback_test COMMAND prefix_writeback_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    add_executable(prefix_hash_affinity_failure_test prefix_hash_affinity_failure_test.cpp "${ROOT}/src/program/prefix_file.cpp")
    target_include_directories(prefix_hash_affinity_failure_test PRIVATE "${ROOT}/include")
    target_link_libraries(prefix_hash_affinity_failure_test PRIVATE Threads::Threads)
    target_link_options(prefix_hash_affinity_failure_test PRIVATE "-Wl,--wrap=pthread_setaffinity_np")
    add_test(NAME prefix_hash_affinity_failure_test COMMAND prefix_hash_affinity_failure_test "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  endif()
  add_executable(prefix_hash_affinity_test prefix_hash_affinity_test.cpp "${ROOT}/src/program/prefix_file.cpp")
  add_executable(prefix_hash_batch_test prefix_hash_batch_test.cpp "${ROOT}/src/program/prefix_file.cpp")
  add_executable(prefix_file_test prefix_file_test.cpp "${ROOT}/src/program/prefix_file.cpp")
  add_executable(prefix_borrow_test prefix_borrow_test.cpp
    "${ROOT}/src/program/prefix_file.cpp" "${ROOT}/src/program/conversation_store.cpp")
  add_executable(conv_cache_test "${ROOT}/src/program/conv_cache_test.cpp")
  add_executable(conv_cache_policy_test conv_cache_policy_test.cpp)
  add_executable(request_stop_test request_stop_test.cpp)
  add_executable(cache_timing_test cache_timing_test.cpp)
  add_executable(cache_resume_plan_test cache_resume_plan_test.cpp)
  target_compile_features(cache_resume_plan_test PRIVATE cxx_std_20)
  add_executable(checkpoint_records_test checkpoint_records_test.cpp)
  add_executable(checkpoint_staging_test checkpoint_staging_test.cpp)
  foreach(target prefix_hash_affinity_test prefix_hash_batch_test prefix_borrow_test cache_timing_test cache_resume_plan_test checkpoint_staging_test checkpoint_records_test conversation_store_test prefix_file_test conv_cache_test conv_cache_policy_test request_stop_test)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    target_include_directories(${target} PRIVATE "${ROOT}/include")
    add_test(NAME ${target} COMMAND ${target} "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
  endforeach()
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  include("${CMAKE_CURRENT_LIST_DIR}/nvfp4_tests.cmake")
endif()
